Penyebab Nomor Tidak Bertambah di Google Sheets:
Google Apps Script di Google Sheets Anda masih menggunakan pemisah strip (-) lama. Karena naskah baru Permenimipas 13/2026 memakai garis miring (/), script di spreadsheet mengabaikan baris 405 dan 406 sehingga nomor tetap terbaca 16.

Langkah 1 Menit untuk Memperbaiki:
  1. Buka spreadsheet Google Sheets Anda.
  2. Klik menu Ekstensi (Extensions) > Apps Script.
  3. Ganti seluruh isi file Code.gs dengan kode di bawah ini.
  4. Klik ikon Simpan (Save).
  5. Klik tombol biru Terapkan (Deploy) > Kelola Penerapan (Manage Deployments) > Edit (Pensil) > Versi Baru (New Version) > Terapkan.

// GOOGLE APPS SCRIPT: REGISTER SURAT KELUAR RUTAN KELAS IIB SINJAI
// Mendukung Pembacaan Nomor Surat Lama (-) dan Format Baru Permenimipas 13/2026 (/)
// ==========================================================================================
var SHEET_NAME = "Sheet1";

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0];
  var data = sheet.getDataRange().getValues();
  var maxNomorPerKode = {};
  
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    var kodeUtama = row[2] ? String(row[2]).trim() : "";
    var klasifikasi = row[3] ? String(row[3]).trim() : "";
    var nomorSurat = row[5] ? String(row[5]).trim() : "";
    if (!nomorSurat) continue;
    
    var angkaSubKode = klasifikasi ? klasifikasi.split(" ")[0].trim() : "";
    var kunci = (kodeUtama && angkaSubKode) ? (kodeUtama + "." + angkaSubKode) : kodeUtama;
    if (!kunci) kunci = cariKodeDariNomor(nomorSurat);
    
    var noUrut = ekstrakNomorUrut(nomorSurat);
    if (kunci && noUrut !== null && !isNaN(noUrut)) {
      if (!maxNomorPerKode[kunci] || noUrut > maxNomorPerKode[kunci]) {
        maxNomorPerKode[kunci] = noUrut;
      }
    }
  }
  
  var hasil = {};
  for (var k in maxNomorPerKode) {
    hasil[k] = maxNomorPerKode[k] + 1;
  }
  return ContentService.createTextOutput(JSON.stringify(hasil)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0];
    var param = e.parameter;
    
    var waktuInput = param.waktu_input || Utilities.formatDate(new Date(), "Asia/Makassar", "yyyy-MM-dd HH:mm:ss");
    var pembuatSurat = param['pembuat-surat'] || "";
    var kodeSurat = param['kode-surat'] || "";
    var klasifikasiSurat = param['klasifikasi-surat'] || "";
    var jenisSurat = param['jenis-surat'] || "";
    var nomorSurat = param['nomor-surat'] || "";
    var tujuanSurat = param['tujuan-surat'] || "";
    var perihal = param.perihal || "";
    var tembusan = param.tembusan || "";
    var tanggalSurat = param['tanggal-surat'] || "";
    
    // Urutan kolom: A=waktuInput, B=pembuat, C=kode, D=klasifikasi, E=jenis, F=nomor, G=tujuan, H=tanggalSurat, I=perihal, J=tembusan
    sheet.appendRow([
      waktuInput, pembuatSurat, kodeSurat, klasifikasiSurat,
      jenisSurat, nomorSurat, tujuanSurat, tanggalSurat, perihal, tembusan
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ "result": "success", "nomor": nomorSurat }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function ekstrakNomorUrut(nomorStr) {
  if (!nomorStr) return null;
  var str = String(nomorStr).trim();
  
  if (str.indexOf("/") !== -1) {
    var parts = str.split("/");
    if (parts.length >= 4 && /^\d+$/.test(parts[1])) return parseInt(parts[1], 10);
    if (parts.length >= 5 && /^\d+$/.test(parts[3])) return parseInt(parts[3], 10);
    if (parts.length >= 4 && /^\d+$/.test(parts[2])) return parseInt(parts[2], 10);
    if (parts.length === 3 && /^\d+$/.test(parts[2])) return parseInt(parts[2], 10);
    for (var i = 1; i < parts.length; i++) {
      if (/^\d+$/.test(parts[i])) {
        var num = parseInt(parts[i], 10);
        if (i !== parts.length - 1 || parts.length <= 3 || num < 1900) return num;
      }
    }
  }
  
  if (str.indexOf("-") !== -1) {
    var segs = str.split("-");
    if (segs.length >= 3 && /^\d+$/.test(segs[1])) return parseInt(segs[1], 10);
    var lastSeg = segs[segs.length - 1].trim();
    if (/^\d+$/.test(lastSeg)) return parseInt(lastSeg, 10);
  }
  return null;
}

function cariKodeDariNomor(nomorStr) {
  if (!nomorStr) return "";
  var match = nomorStr.match(/([A-Z]{2}\.\d{2}(\.\d{2})?)/);
  return match ? match[1] : "";
}
